"""A5-900 credential and authority-grant scan.

Scans this node's ENTIRE write scope for (a) anything credential-shaped and
(b) any text that would grant, request, or embed credential, spend, merge, or
production authority.

It never prints a matched secret value: a credential-shaped hit is reported as
path + line number + pattern name only. Nothing in this scanner reads any
environment variable; the token-name patterns are literal substrings used to
detect a NAME appearing in evidence, not to fetch a value.
"""

from __future__ import annotations

import re
import sys
from pathlib import Path

ROOT = Path(r"C:/Repos/HiveMind/hive-mind-os")
TARGETS = [
    ROOT / "evidence" / "pilots" / "a5",
    ROOT / "docs" / "execution" / "A5_GOVERNED_FULL_READINESS.md",
]

# (a) credential-shaped. A hit here is a FAILURE.
SECRET_PATTERNS = {
    "github-classic-pat": re.compile(r"gh[pousr]_[A-Za-z0-9]{16,}"),
    "github-fine-grained-pat": re.compile(r"github_pat_[A-Za-z0-9_]{20,}"),
    "aws-access-key": re.compile(r"AKIA[0-9A-Z]{16}"),
    "slack-token": re.compile(r"xox[abprs]-[A-Za-z0-9-]{10,}"),
    "private-key-header": re.compile(r"-----BEGIN [A-Z ]*PRIVATE KEY-----"),
    "bearer-with-value": re.compile(r"Bearer\s+[A-Za-z0-9._~+/=-]{12,}"),
    "authorization-header-value": re.compile(r"[Aa]uthorization\"?\s*[:=]\s*\"?[A-Za-z0-9._~+/=-]{12,}"),
    # '/' is deliberately excluded: filesystem paths in transcripts are otherwise
    # indistinguishable from base64 runs and produced only false positives.
    "base64ish-run-40plus": re.compile(r"(?<![A-Za-z0-9+/=_-])[A-Za-z0-9+]{40,}={0,2}(?![A-Za-z0-9+/=_-])"),
    "jwt": re.compile(r"eyJ[A-Za-z0-9_-]{10,}\.[A-Za-z0-9_-]{10,}\."),
}
# sha256 hex digests are legitimate evidence and are excluded from the entropy rule.
SHA256_HEX = re.compile(r"\b[0-9a-f]{64}\b")
GIT_SHA = re.compile(r"\b[0-9a-f]{7,40}\b")

# (b) token variable NAMES appearing in evidence. A hit is reported for review.
NAME_PATTERNS = {
    "token-env-var-name-fragment": re.compile(r"[A-Z0-9_]*PILOT_TOKEN"),
    "github-token-env-var-name": re.compile(r"GITHUB_TOKEN"),
    "generic-secret-word": re.compile(r"(?i)\b(password|passphrase|api[_-]?key|client[_-]?secret)\b"),
}

# (c) authority-granting or authority-requesting language. A hit is adjudicated.
GRANT_PATTERNS = {
    "grants-authority": re.compile(r"(?i)\b(this (record|document|node|evidence) (grants|authorizes|permits|approves))\b"),
    "requests-credential": re.compile(r"(?i)\b(please (provide|paste|supply|export) (a |the )?(token|credential|key|password))\b"),
    "declares-ready": re.compile(r"(?i)(\bverdict\W+ready\b|\bproduction[- ]ready\b|\brelease[- ]ready\W+true\b|\bis now ready\b)"),
    "superiority": re.compile(r"(?i)\b(better than|outperform\w*|superior to|state[- ]of[- ]the[- ]art|best[- ]in[- ]class)\b"),
    "gate-softening": re.compile(r"(?i)\b(effectively (open|satisfied)|can be treated as (open|satisfied)|de facto (open|granted)|waive[ds]? the gate)\b"),
}


def files() -> list[Path]:
    out: list[Path] = []
    for target in TARGETS:
        if target.is_dir():
            out.extend(sorted(p for p in target.rglob("*") if p.is_file()))
        elif target.is_file():
            out.append(target)
        else:
            print(f"  (target absent, not scanned: {target.relative_to(ROOT).as_posix()})")
    return out


def main() -> int:
    scanned = files()
    print(f"files_scanned={len(scanned)}")
    for path in scanned:
        print(f"  {path.relative_to(ROOT).as_posix()}")
    print()

    failures = 0
    for label, group, is_failure in (
        ("A. CREDENTIAL-SHAPED (any hit is a failure)", SECRET_PATTERNS, True),
        ("B. TOKEN VARIABLE NAMES AND SECRET WORDS (reported for review)", NAME_PATTERNS, False),
        ("C. AUTHORITY-GRANTING / GATE-SOFTENING / SUPERIORITY LANGUAGE (adjudicated)", GRANT_PATTERNS, False),
    ):
        print(f"== {label} ==")
        hits = 0
        for path in scanned:
            try:
                text = path.read_text(encoding="utf-8", errors="replace")
            except OSError as error:
                print(f"  UNREADABLE {path}: {error}")
                continue
            for number, line in enumerate(text.splitlines(), start=1):
                for name, pattern in group.items():
                    match = pattern.search(line)
                    if match is None:
                        continue
                    if name == "base64ish-run-40plus":
                        span = match.group(0)
                        if SHA256_HEX.fullmatch(span) or GIT_SHA.fullmatch(span):
                            continue
                        if set(span) <= set("0123456789abcdef"):
                            continue
                    hits += 1
                    if is_failure:
                        failures += 1
                        print(f"  HIT {path.relative_to(ROOT).as_posix()}:{number} pattern={name} (value withheld)")
                    else:
                        excerpt = line.strip()
                        excerpt = excerpt[:150] + ("..." if len(excerpt) > 150 else "")
                        print(f"  HIT {path.relative_to(ROOT).as_posix()}:{number} pattern={name}")
                        print(f"      {excerpt}")
        if hits == 0:
            print("  CLEAN: 0 hits")
        print()

    print(f"credential_shaped_failures={failures}")
    print("RESULT:", "PASS -- no credential-shaped content" if failures == 0 else "FAIL")
    return 0


if __name__ == "__main__":
    sys.exit(main())
