"""A5-900 stop-mechanism exercise driver.

A four-step, receipt-ledgered procedure over RETAINED A4 evidence. Every step is
a read-only digest measurement -- the procedure creates no effect outside its own
ledger directory and touches no network and no credential.

Modes:
  --stop-after N          the stop trigger fires after step N: issue no further
                          step, write the stop record, exit 0
  --kill-before-receipt N perform step N's effect, then die HARD (os._exit) before
                          the receipt is appended -- the shape of open defect D1
  (default)               resume: adopt every step that already has a receipt,
                          execute only the rest
"""

from __future__ import annotations

import argparse
import hashlib
import json
import os
import sys
from pathlib import Path

REPO = Path(r"C:/Repos/HiveMind/hive-mind-os")

STEPS = [
    ("S1", "digest evidence/pilots/a4/pilot-plan.json", "evidence/pilots/a4/pilot-plan.json"),
    ("S2", "digest evidence/pilots/a4/summary.json", "evidence/pilots/a4/summary.json"),
    ("S3", "digest evidence/pilots/a4/owner-grant/gate-stop.json", "evidence/pilots/a4/owner-grant/gate-stop.json"),
    ("S4", "digest evidence/pilots/a4/owner-grant/grant-record.json", "evidence/pilots/a4/owner-grant/grant-record.json"),
]


def sha256(path: Path) -> str:
    return hashlib.sha256(path.read_bytes()).hexdigest()


def load_ledger(ledger: Path) -> dict[str, dict]:
    if not ledger.exists():
        return {}
    adopted: dict[str, dict] = {}
    for line in ledger.read_text(encoding="utf-8").splitlines():
        line = line.strip()
        if not line:
            continue
        record = json.loads(line)
        if record["step"] in adopted:
            print(f"  !! DUPLICATE RECEIPT for {record['step']} -- this would be a defect")
        adopted[record["step"]] = record
    return adopted


def append_receipt(ledger: Path, record: dict) -> None:
    with ledger.open("a", encoding="utf-8", newline="\n") as handle:
        handle.write(json.dumps(record, sort_keys=True) + "\n")
        handle.flush()
        os.fsync(handle.fileno())


def main() -> int:
    parser = argparse.ArgumentParser()
    parser.add_argument("--dir", required=True)
    parser.add_argument("--run", required=True)
    parser.add_argument("--stop-after", type=int, default=0)
    parser.add_argument("--kill-before-receipt", type=int, default=0)
    args = parser.parse_args()

    root = Path(args.dir)
    root.mkdir(parents=True, exist_ok=True)
    ledger = root / "ledger.jsonl"

    adopted = load_ledger(ledger)
    print(f"-- run {args.run}: ledger holds {len(adopted)} receipt(s): {sorted(adopted)}")

    executed: list[str] = []
    for index, (step, label, relative) in enumerate(STEPS, start=1):
        if step in adopted:
            print(f"  [{step}] ADOPTED from ledger (not re-executed): {adopted[step]['digest'][:16]}...")
            continue

        digest = sha256(REPO / relative)
        print(f"  [{step}] EXECUTED {label} -> {digest[:16]}...")

        if args.kill_before_receipt == index:
            print(f"  [{step}] HARD KILL before the receipt is appended (open defect D1 shape)")
            sys.stdout.flush()
            os._exit(9)

        append_receipt(ledger, {"step": step, "label": label, "path": relative,
                                "digest": digest, "run": args.run})
        executed.append(step)

        if args.stop_after == index:
            stop = {"stopped_after": step, "run": args.run,
                    "reason": "stop trigger fired: issue no further step",
                    "receipts_at_stop": sorted(load_ledger(ledger))}
            (root / f"stop-record-{args.run}.json").write_text(
                json.dumps(stop, indent=2, sort_keys=True) + "\n", encoding="utf-8")
            print(f"  STOP: trigger fired after {step}. Wrote stop-record-{args.run}.json")
            print(f"  executed_this_run={executed}")
            return 0

    print(f"  executed_this_run={executed}")
    print(f"  completed: {len(load_ledger(ledger))}/{len(STEPS)} steps have receipts")
    return 0


if __name__ == "__main__":
    raise SystemExit(main())
