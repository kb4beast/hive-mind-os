def slot_forge(envelope_digest, action, target, witness=""):
    """Build a CapabilityToken bypassing __post_init__ entirely.

    NEW. The retained probe's hand-built token now fails at CONSTRUCTION, which
    is a refusal but not a measurement of the effect boundary. This forges the
    object at the slot level so the boundary itself is what has to refuse it --
    the strongest in-process forgery short of reading the issuance key.
    """

    token = object.__new__(CapabilityToken)
    object.__setattr__(token, "envelope_digest", envelope_digest)
    object.__setattr__(token, "action", action)
    object.__setattr__(token, "target", target)
    object.__setattr__(
        token,
        "token_digest",
        canonical_digest({"envelope": envelope_digest, "action": action, "target": target}),
    )
    object.__setattr__(token, "issuance_witness", witness)
    return token


def report(label, fn, expect_denied):
    try:
        result = fn()
    except AuthorityDenied as error:
        outcome, detail = "DENIED", f"AuthorityDenied: {error}"
    except Exception as error:  # noqa: BLE001 - a crash must not read as a refusal
        outcome, detail = "CRASHED", f"{type(error).__name__}: {error}"
    else:
        outcome, detail = "ALLOWED", repr(result)
    verdict = "as-expected" if (outcome == "DENIED") == expect_denied else "FINDING"
    print(f"[{verdict:12}] {label}")
    print(f"               expected={'DENIED' if expect_denied else 'ALLOWED'} observed={outcome}")
    print(f"               detail={detail}")
    return outcome
