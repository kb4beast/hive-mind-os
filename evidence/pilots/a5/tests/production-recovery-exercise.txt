#!/usr/bin/env bash
# A5-900 production-recovery exercise (retained run). Local only.
# No network, no credential, no production system. Each recovery uses a FRESH
# destination directory so no measurement is confounded by deleting a prior one.
set -u

REPO="C:/Repos/HiveMind/hive-mind-os"
SCR="C:/Users/beesp/AppData/Local/Temp/claude/C--Repos-HiveMind-hive-mind-os/7a67801b-465b-4082-948f-c38656749678/scratchpad/a5"
SHORT="C:/Repos/trash/a5rt"
BRANCH="release/hive-mind-os-singleton-20260812-r5"
ARTIFACT="evidence/pilots/a4/pilot-plan.json"
RECORDED="324151468418014ed70c22162d2689d3ab7d9ac1770e80fc7ca4b228fc9a9fbc"

step() { echo; echo "### $1"; echo "\$ $2"; }
do_step() { step "$1" "$2"; eval "$2" 2>&1 | tail -"${3:-12}"; echo "[exit_code=${PIPESTATUS[0]}]"; }

# Count tracked paths that exist on disk, using a longpaths-aware ls-files.
materialised() {
  local root="$1" miss=0 total=0 f
  while IFS= read -r f; do
    total=$((total+1)); [ -f "$root/$f" ] || miss=$((miss+1))
  done < <(git -C "$root" -c core.longpaths=true ls-files)
  echo "tracked=$total missing_on_disk=$miss"
}

echo "===== A5-900 production-recovery exercise ====="
echo "node: A5-900"
echo "utc_start: $(date -u +%Y-%m-%dT%H:%M:%SZ)"
echo "host: $(uname -s) $(uname -r)"
echo "git: $(git --version)"
echo "source_repo: $REPO"
echo "network: NONE. Every command below is local filesystem or local git."
echo "credential: NONE read, referenced, or required."

echo
echo "===== R0 baseline ====="
do_step "R0.1 source HEAD" "git -C '$REPO' rev-parse HEAD"
do_step "R0.2 source HEAD tree" "git -C '$REPO' rev-parse HEAD^{tree}"
do_step "R0.3 source branch" "git -C '$REPO' rev-parse --abbrev-ref HEAD"
do_step "R0.4 digest of record for $ARTIFACT (evidence/pilots/a4/summary.json file_digests)" "echo $RECORDED"
do_step "R0.5 on-disk digest of that artifact in the source repo" "sha256sum '$REPO/$ARTIFACT'"
do_step "R0.6 longest tracked path length" "git -C '$REPO' ls-files | awk '{print length(\$0)}' | sort -rn | head -1"
do_step "R0.7 core.longpaths in the SOURCE repo (local / global / system)" "echo local=\$(git -C '$REPO' config --local core.longpaths || echo UNSET); echo global=\$(git -C '$REPO' config --global core.longpaths || echo UNSET); echo system=\$(git -C '$REPO' config --system core.longpaths || echo UNSET)"

echo
echo "===== R1 fresh clone at a normal temp path, DEFAULT git config (the naive recovery) ====="
R1="$SCR/rec1"
rm -rf "$R1" 2>/dev/null
step "R1.1 git clone (default config), destination length ${#R1}" "git clone '$REPO' '$R1'"
git clone "$REPO" "$R1" > "$SCR/rec1.txt" 2>&1; R1E=$?
echo "[exit_code=$R1E]"
echo "filename_too_long_errors=$(grep -c 'Filename too long' "$SCR/rec1.txt")"
grep 'Filename too long' "$SCR/rec1.txt" | head -1 | cut -c1-200
tail -3 "$SCR/rec1.txt"
do_step "R1.2 objects DID transfer: clone HEAD" "git -C '$R1' rev-parse HEAD"
do_step "R1.3 index deletions reported by git status (first-column 'D')" "git -C '$R1' status --porcelain | grep -c '^D '"
do_step "R1.4 materialisation census" "materialised '$R1'"
do_step "R1.5 VERDICT R1" "echo 'clone exit 128; history recovered, working tree NOT recovered; this repository cannot be recovered at this path with default git config'"

echo
echo "===== R2 fresh clone at a SHORT path, DEFAULT git config ====="
R2="$SHORT/rec2"
rm -rf "$R2" 2>/dev/null; mkdir -p "$SHORT"
step "R2.1 git clone (default config), destination length ${#R2}" "git clone '$REPO' '$R2'"
git clone "$REPO" "$R2" > "$SCR/rec2.txt" 2>&1; R2E=$?
echo "[exit_code=$R2E]"
echo "filename_too_long_errors=$(grep -c 'Filename too long' "$SCR/rec2.txt")"
tail -2 "$SCR/rec2.txt"
