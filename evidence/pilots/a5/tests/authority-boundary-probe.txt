D1 = "sha256:" + "1" * 64

def envelope(digest, parent, allowed, denied, write_scope):
    return ConstraintEnvelope(
        "AUTH-probe",
        "MISSION-probe",
        "WORK-probe",
        parent,
        "builder",
        "R1",
        allowed,
        denied,
        write_scope,
        write_scope,
        (),
        (),
        (),
        (),
        Budget(1, 0, 0, 0, 0, 0, 1, 1),
        "2030-01-01T00:00:00Z",
        POLICY,
        digest,
    )


def check(label, fn, expect_denied):
    try:
        result = fn()
    except (AuthorityDenied, DeliveryGrantError) as error:
        outcome = "DENIED"
        detail = f"{type(error).__name__}: {error}"
    else:
        outcome = "ALLOWED"
        detail = repr(result)
    verdict = "as-expected" if (outcome == "DENIED") == expect_denied else "FINDING"
    print(f"[{verdict:12}] {label}")
    print(f"               expected={'DENIED' if expect_denied else 'ALLOWED'} observed={outcome}")
    print(f"               detail={detail}")
    return outcome
