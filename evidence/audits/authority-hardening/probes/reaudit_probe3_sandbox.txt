"""REAUDIT-1900 residual measurement: how load-bearing is the containment proof?

Residual 5 as declared by SANDBOX-1300: "the containment proof is load-bearing on

This measures it WITHOUT editing tests/ or src/ (both forbidden here): it runs

from __future__ import annotations

import os
import sys
import unittest
from pathlib import Path

REPO = r"C:\Repos\HiveMind\hive-mind-os"
sys.path.insert(0, REPO)

import hive_mind_os.sandbox as sandbox_module

SANDBOX_FILE = os.path.abspath(sandbox_module.__file__)

# Locate the guard by content rather than by a hard-coded line number.
lines = Path(SANDBOX_FILE).read_text(encoding="utf-8").splitlines()
GUARD_LINES = {
    n
    for n, text in enumerate(lines, start=1)
    if "resolved.relative_to(self.spec.root)" in text
    or "path argument escapes sandbox root" in text
}
print(f"sandbox module      : {SANDBOX_FILE}")
print(f"guard line numbers  : {sorted(GUARD_LINES)}")
for n in sorted(GUARD_LINES):
    print(f"  {n}: {lines[n - 1].strip()}")
print()

hits: dict[int, set[str]] = {n: set() for n in GUARD_LINES}
current = {"test": "<none>"}


def tracer(frame, event, arg):
    if event == "call":
        return tracer if os.path.abspath(frame.f_code.co_filename) == SANDBOX_FILE else None
    if event == "line" and frame.f_lineno in hits:
        hits[frame.f_lineno].add(current["test"])
    return tracer


class Recorder(unittest.TextTestResult):
    def startTest(self, test):
        current["test"] = test.id().split(".")[-1]
        super().startTest(test)


loader = unittest.TestLoader()
suite = loader.loadTestsFromName("tests.test_sandbox")
sys.settrace(tracer)
threading_ok = True
try:
    import threading

    threading.settrace(tracer)
except Exception:  # noqa: BLE001
    threading_ok = False
runner = unittest.TextTestRunner(verbosity=0, resultclass=Recorder, stream=sys.stderr)
result = runner.run(suite)
sys.settrace(None)

print()
print(f"tests.test_sandbox  : Ran {result.testsRun}, failures={len(result.failures)}, "
      f"errors={len(result.errors)}, skipped={len(result.skipped)}")
print(f"threading trace set : {threading_ok}")
print()
print("which tests execute each guard line:")
for n in sorted(GUARD_LINES):
    names = sorted(hits[n])
    print(f"  line {n}: {len(names)} test(s) -> {names or '<NONE>'}")
print()
refusal = [n for n in sorted(GUARD_LINES) if "escapes sandbox root" in lines[n - 1]]
for n in refusal:
